// TERMINAL UTAMA (ROUTER + SISTEM KEAMANAN TOKEN)
// =================================================================
function doPost(e) {
  try {
    var action = e.parameter.action;

    // 1. IZINKAN AKSES LOGIN & PORTAL ORTU TANPA TOKEN
if (action === 'login') return prosesLogin(e);
if (action === 'simpanAbsenBarcode') return simpanAbsenBarcode(e);

// --- JALUR KHUSUS PORTAL WALI SANTRI ---
    if (action === 'getSppSantri') return getSppSantri(e);
    if (action === 'getRaporTK') return getRaporTK(e);
    
    // TAMBAHAN: Izinkan Wali Santri menarik data ini tanpa Token Login
    if (action === 'getSantri') return ambilDataSantri();
    if (action === 'getAllMapel') return getAllMapel();
    if (action === 'getDataNilai') return getDataNilai(e);
    if (action === 'getPengaturan') return getPengaturan(e);
    if (action === 'getRankingKelas') return getRankingKelas(e);
    if (action === 'simpanAbsenBarcode') return simpanAbsenBarcode(e);
    if (action === 'getRekapBulanan') return getRekapBulanan(e);
    if (action === 'getBulanTersedia') return getBulanTersedia();
    if (action === 'getSettingSpp') return getSettingSpp();
    

// (Jika nanti ortu.html memanggil fungsi lain, tambahkan juga di sini)

    // ==========================================
    // 2. AREA SATPAM: VALIDASI TOKEN (SESI)
    // ==========================================
    var tokenKlien = e.parameter.token;
    
    if (!tokenKlien) {
      return responError('Akses Ditolak: Anda belum login atau sesi telah berakhir.');
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetSesi = ss.getSheetByName('SesiAktif');
    
    // Auto-Create Sheet SesiAktif jika belum ada
    if (!sheetSesi) {
      sheetSesi = ss.insertSheet('SesiAktif');
      sheetSesi.appendRow(['Token', 'Username', 'Role', 'Waktu Login']);
      sheetSesi.getRange("A1:D1").setFontWeight("bold").setBackground("#f87171"); // Warna merah sbg tanda file sistem
    }

    var dataSesi = sheetSesi.getDataRange().getValues();
    var tokenValid = false;
    var userRole = "";

    // Pengecekan apakah token ada di database
    for (var i = 1; i < dataSesi.length; i++) {
      if (dataSesi[i][0] === tokenKlien) {
        tokenValid = true;
        userRole = dataSesi[i][2];
        break;
      }
    }

    // Jika token palsu/tidak ada di database, tendang!
    if (!tokenValid) {
      return responError('Sesi tidak valid atau telah kadaluarsa. Silakan login ulang.');
    }
    // ==========================================
    // AREA SATPAM SELESAI
    // ==========================================


    // ==========================================
    // 3. JIKA TOKEN VALID, LANJUTKAN KE ROUTER
    // ==========================================
    
    // (Opsional) Keamanan Ekstra: Kunci fitur khusus hanya untuk Admin
    var adminOnly = ['mutasiSantri', 'importSantriBulk', 'deleteSppData', 'updateDataNilai', 'deleteSantri'];
    var roleBersih = userRole.toString().trim(); // Bersihkan spasi gaib jika ada
    
    // Tambahkan 'Administrator' ke dalam daftar putih (whitelist)
    if (adminOnly.indexOf(action) !== -1 && roleBersih !== 'Admin' && roleBersih !== 'Administrator' && roleBersih !== 'Ustadz Admin Utama') {
        return responError('Akses Ditolak: Anda tidak memiliki hak akses Admin untuk fitur ini!');
    }

    switch (action) {
      case 'getSantri': return ambilDataSantri();
      case 'addSantri': return tambahDataSantri(e);
      case 'updateSantri': return updateDataSantri(e);
      case 'deleteSantri': return hapusDataSantriServer(e);
      case 'simpanNilai': return simpanDataNilai(e);
      case 'getDataNilai': return getDataNilai(e); 
      case 'updateDataNilai': return updateDataNilai(e);
      case 'getRankingKelas': return getRankingKelas(e);
      case 'getBintangPelajar': return getBintangPelajar();
      case 'getRaporTK': return getRaporTK(e);
      case 'getPengaturan': return getPengaturan(e);
      case 'simpanPengaturan': return simpanPengaturan(e);
      case 'uploadGambar': return uploadGambarDrive(e);
      case 'getAllMapel': return getAllMapel();
      case 'notifLogin': return kirimNotifLogin(e);
      case 'mutasiSantri': return mutasiSantri(e);
      case 'importSantriBulk': return importSantriBulk(e);
      case 'getSppData': return getSppData(e);
      case 'saveSppData': return saveSppData(e);
      case 'deleteSppData': return deleteSppData(e);
      case 'getSppSantri': return getSppSantri(e); 
      case 'toggleStatusAbsen': return toggleStatusAbsen(e);
      case 'getStatusAbsen': return getStatusAbsen();

      case 'getSettingSpp': return getSettingSpp();
      case 'saveSettingSpp': return saveSettingSpp(e);

      case 'getBukuKas': return getBukuKas();
      case 'addPengeluaran': return addPengeluaran(e);
      case 'getLaporanKas': return getLaporanKas(e);
      case 'deleteKas': return deleteKas(e);
      
      // --- KODE BARU: RUTE UNTUK FITUR ABSENSI ---
      case 'setLiburRentang': return setLiburRentang(e);
      case 'simpanAbsenBarcode': return simpanAbsenBarcode(e);
      case 'getBulanTersedia': return getBulanTersedia(e);
      case 'getRekapBulanan': return getRekapBulanan(e);
      case 'simpanWaktuAbsen': return simpanWaktuAbsen(e);
      case 'simpanLokasiAbsen': return simpanLokasiAbsen(e);
      case 'hapusLokasiAbsen': return hapusLokasiAbsen(e);
      
      default: return responError('Aksi tidak dikenali oleh server.');
    }
  } catch (error) {
    return responError('Terjadi kesalahan server: ' + error.message);
  }
}

// 1. FUNGSI LOGIN (PEMBUAT TOKEN)
// =================================================================
function prosesLogin(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('LOGIN');
  
  if (!sheet) return responError('Sistem Error: Sheet "LOGIN" tidak ditemukan di database.');

  // Bersihkan spasi kosong jika ada
  var user = e.parameter.username ? e.parameter.username.toString().trim() : "";
  var pass = e.parameter.password ? e.parameter.password.toString().trim() : "";
  var data = sheet.getDataRange().getValues();

  for (var i = 1; i < data.length; i++) {
    // Cocokkan data (pastikan huruf besar/kecil dan spasi akurat)
    if (data[i][0].toString().trim() === user && data[i][1].toString().trim() === pass) {
      var roleUser = data[i][2];
      var namaUser = data[i][3];

      // --- KODE BARU: PEMBUATAN TOKEN ---
      var token = Utilities.getUuid();
      var sheetSesi = ss.getSheetByName('SesiAktif');
      
      // Buat sheet jika belum ada
      if (!sheetSesi) {
        sheetSesi = ss.insertSheet('SesiAktif');
        sheetSesi.appendRow(['Token', 'Username', 'Role', 'Waktu Login']);
        sheetSesi.getRange("A1:D1").setFontWeight("bold").setBackground("#f87171");
      }

      // Simpan token ke database SesiAktif
      sheetSesi.appendRow([token, user, roleUser, new Date()]);

      // Kirim balik data ke browser beserta Token-nya
      return responSukses({ 
        role: roleUser, 
        name: namaUser,
        token: token 
      });
    }
  }
  return responError('Username atau Password salah! Periksa kembali data Anda.');
}

// 2. FUNGSI AMBIL DATA SANTRI
// =================================================================
function ambilDataSantri() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('SANTRI');
  var data = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < data.length; i++) {
    result.push({
      nis: data[i][0], 
      nama: data[i][1], 
      jk: data[i][2], 
      ttl: data[i][3] || '',     // <-- POSISI BARU (Indeks 3)
      kelas: data[i][4],         // <-- Bergeser jadi indeks 4
      alamat: data[i][5] || '',  // <-- Bergeser jadi indeks 5
      ayah: data[i][6] || '',    // <-- Bergeser jadi indeks 6
      ibu: data[i][7] || '',     // <-- Bergeser jadi indeks 7
      hp: data[i][8] || ''       // <-- Bergeser jadi indeks 8
    });
  }
  return responSukses({ data: result });
}

// =================================================================
// 3. FUNGSI TAMBAH DATA SANTRI 
// =================================================================
function tambahDataSantri(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('SANTRI');
  var inputNis = e.parameter.nis;
  var existingData = sheet.getDataRange().getValues();
  
  for (var i = 1; i < existingData.length; i++) {
    if (existingData[i][0] == inputNis) return responError('Gagal! NIS ' + inputNis + ' sudah terdaftar.');
  }
  
  var saveNis = "'" + inputNis;
  var saveHp = "'" + e.parameter.hp;
  
  // <-- POSISI BARU: e.parameter.ttl ditaruh setelah jk
  sheet.appendRow([
    saveNis, e.parameter.nama, e.parameter.jk, e.parameter.ttl, 
    e.parameter.kelas, e.parameter.alamat, e.parameter.ayah, e.parameter.ibu, saveHp
  ]);
  
  return responSukses({ message: 'Data santri berhasil ditambahkan!' });
}

// 4. FUNGSI UPDATE/EDIT DATA SANTRI (DIPERBARUI UNTUK EDIT NIS)
// =================================================================
function updateDataSantri(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('SANTRI');
  var inputNisLama = e.parameter.nis_lama; // NIS asli untuk mencari data
  var inputNisBaru = e.parameter.nis;      // NIS baru hasil editan
  
  var data = sheet.getDataRange().getValues();
  var isUpdated = false;
  var saveHp = "'" + e.parameter.hp;
  var saveNisBaru = "'" + inputNisBaru;

  for (var i = 1; i < data.length; i++) {
    // Cari baris berdasarkan NIS Lama
    if (data[i][0] == inputNisLama) {
      sheet.getRange(i + 1, 1).setValue(saveNisBaru);        // <-- Update NIS (Kolom 1)
      sheet.getRange(i + 1, 2).setValue(e.parameter.nama);
      sheet.getRange(i + 1, 3).setValue(e.parameter.jk);
      sheet.getRange(i + 1, 4).setValue(e.parameter.ttl);    
      sheet.getRange(i + 1, 5).setValue(e.parameter.kelas);  
      sheet.getRange(i + 1, 6).setValue(e.parameter.alamat); 
      sheet.getRange(i + 1, 7).setValue(e.parameter.ayah);   
      sheet.getRange(i + 1, 8).setValue(e.parameter.ibu);    
      sheet.getRange(i + 1, 9).setValue(saveHp);             
      isUpdated = true; break;
    }
  }
  if (isUpdated) return responSukses({ message: 'Data santri berhasil diperbarui!' });
  else return responError('Gagal: Data santri tidak ditemukan.');
}

// FUNGSI MUTASI KELAS MASSAL (OPTIMASI SKALA BESAR)
// =================================================================
function mutasiSantri(e) {
  // 1. Terapkan LockService (Lampu Lalu Lintas)
  // Mencegah data bentrok jika ada 2 admin melakukan mutasi di detik yang sama
  var lock = LockService.getScriptLock();
  
  // Tunggu maksimal 10 detik. Jika masih ada proses lain, tolak dengan halus.
  if (!lock.tryLock(10000)) {
    return responError('Sistem sedang sibuk memproses data lain. Silakan coba beberapa saat lagi.');
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var kelasTujuan = e.parameter.kelas_tujuan;
    var nisList = JSON.parse(e.parameter.nis_list);

    var sheetSantri = ss.getSheetByName('SANTRI');
    var lastRow = sheetSantri.getLastRow();
    
    // Cegah error jika sheet hanya berisi header
    if (lastRow < 2) {
      return responError('Tidak ada data santri untuk dimutasi.');
    }

    // 2. Tarik hanya kolom yang dibutuhkan (Kolom 1: NIS, Kolom 5: Kelas) ke dalam memori
    // Membaca area spesifik jauh lebih ringan daripada getDataRange() untuk 1000 baris
    var dataNis = sheetSantri.getRange(2, 1, lastRow - 1, 1).getValues();
    var dataKelas = sheetSantri.getRange(2, 5, lastRow - 1, 1).getValues();
    
    var jumlahDiupdate = 0;

    // 3. Proses pencocokan di dalam memori lokal (Sangat Cepat)
    for (var i = 0; i < dataNis.length; i++) {
      var nisBaris = dataNis[i][0].toString().replace(/'/g, "").trim();

      // Jika NIS di baris ini ada di dalam daftar centangan admin
      if (nisList.indexOf(nisBaris) !== -1) {
        dataKelas[i][0] = kelasTujuan; // Ubah data di dalam memori array
        jumlahDiupdate++;
      }
    }

    // 4. BATCH PROCESSING: Tulis kembali seluruh perubahan ke Sheet dalam 1X Perintah
    if (jumlahDiupdate > 0) {
      sheetSantri.getRange(2, 5, dataKelas.length, 1).setValues(dataKelas);
    }

    return responSukses({
      message: jumlahDiupdate + ' santri berhasil dimutasi ke ' + kelasTujuan
    });

  } catch (error) {
    return responError('Terjadi kesalahan sistem: ' + error.message);
  } finally {
    // 5. Wajib lepaskan kunci agar admin lain bisa menggunakan sistem kembali
    lock.releaseLock();
  }
}

// 5. FUNGSI SIMPAN BUKU NILAI MASSAL (OPTIMASI BATCH & LOCK)
// =================================================================
function simpanDataNilai(e) {
  // 1. Lampu Lalu Lintas: Antre maksimal 15 detik jika ada guru lain yang sedang menyimpan
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    return responError('Server sedang sibuk memproses nilai dari guru lain. Silakan coba klik simpan lagi dalam beberapa detik.');
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var kelasPilih = e.parameter.kelas;
    var listNilai = JSON.parse(e.parameter.list_nilai);
    var sheetName = 'NILAI ' + kelasPilih;
    var sheet = ss.getSheetByName(sheetName);

    // ==========================================
    // LOGIKA KHUSUS TK / RA
    // ==========================================
    if (kelasPilih.includes('TK')) {
      var hari = e.parameter.hari;
      if (!sheet) {
        sheet = ss.insertSheet(sheetName);
        sheet.appendRow(['NIS', 'Nama Lengkap', 'Kelas', 'Hari', 'Total Nilai', 'Rata-rata', 'Mapel 1', 'Nilai 1', 'Mapel 2', 'Nilai 2']);
        sheet.getRange(1, 1, 1, 10).setFontWeight("bold").setBackground("#d1fae5");
      }

      var dataTK = sheet.getDataRange().getValues();
      
      // Validasi Anti-Duplikat di Memori (Cepat)
      for (var n = 0; n < listNilai.length; n++) {
        for (var i = 1; i < dataTK.length; i++) {
          if (dataTK[i][0].toString() === listNilai[n].nis.toString() && dataTK[i][3].toString() === hari.toString()) {
            return responError('GAGAL: Santri bernama ' + listNilai[n].nama + ' sudah pernah dinilai di hari ' + hari);
          }
        }
      }

      // Siapkan Array untuk Batch Insert
      var barisBaruTK = [];
      for (var n = 0; n < listNilai.length; n++) {
        barisBaruTK.push([
          "'" + listNilai[n].nis, listNilai[n].nama, kelasPilih, hari, 
          listNilai[n].total, listNilai[n].rata, 
          listNilai[n].m1, listNilai[n].n1, 
          listNilai[n].m2, listNilai[n].n2
        ]);
      }

      // Tulis Sekaligus ke Database
      if (barisBaruTK.length > 0) {
        sheet.getRange(sheet.getLastRow() + 1, 1, barisBaruTK.length, barisBaruTK[0].length).setValues(barisBaruTK);
      }
      return responSukses({ message: 'Nilai TK berhasil disimpan secara massal.' });
    } 
    
    // ==========================================
    // LOGIKA IBTIDAIYAH, SANAWIYAH, ALIYAH
    // ==========================================
    else {
      var mapel = e.parameter.mapel;
      var semuaMapel = JSON.parse(e.parameter.semua_mapel);
      var headersBase = ['NIS', 'Nama Lengkap', 'Kelas', 'Total Nilai', 'Rata-rata'];
      var fullHeaders = headersBase.concat(semuaMapel);
      
      if (!sheet) {
        sheet = ss.insertSheet(sheetName);
        sheet.appendRow(fullHeaders);
        sheet.getRange(1, 1, 1, fullHeaders.length).setFontWeight("bold").setBackground("#d1fae5");
      }

      var data = sheet.getDataRange().getValues();
      var sheetHeaders = data[0];
      var colIndex = sheetHeaders.indexOf(mapel);
      
      // Petakan index baris berdasarkan NIS agar pencarian O(1) sangat cepat
      var mapExisting = {};
      for (var i = 1; i < data.length; i++) {
        var nisData = data[i][0].toString().replace(/'/g, "").trim();
        mapExisting[nisData] = i; 
      }

      var barisBaru = [];

      // Kalkulasi seluruh nilai di dalam Memori
      for (var n = 0; n < listNilai.length; n++) {
        var student = listNilai[n];
        var nisStudent = student.nis.toString().replace(/'/g, "").trim();
        var existingIndex = mapExisting[nisStudent];

        if (existingIndex !== undefined) {
          // UPDATE DATA LAMA DI MEMORI
          data[existingIndex][colIndex] = student.nilai;

          // Hitung ulang total dan rata-rata secara langsung
          var total = 0; 
          var totalMapel = sheetHeaders.length - 5;
          for (var h = 5; h < sheetHeaders.length; h++) {
            var val = data[existingIndex][h];
            if (val !== "" && !isNaN(val)) { total += parseFloat(val); }
          }
          var rata = totalMapel > 0 ? (total / totalMapel).toFixed(1) : 0;

          data[existingIndex][3] = total; // Index 3: Total Kolom
          data[existingIndex][4] = rata;  // Index 4: Rata-rata Kolom

        } else {
          // JIKA ADA SANTRI BARU SUSULAN (Belum ada di sheet)
          var totalMapel = sheetHeaders.length - 5;
          var rataAwal = totalMapel > 0 ? (parseFloat(student.nilai) / totalMapel).toFixed(1) : student.nilai;
          var newRow = ["'" + student.nis, student.nama, kelasPilih, student.nilai, rataAwal];
          
          for (var h = 5; h < sheetHeaders.length; h++) {
            if (sheetHeaders[h] === mapel) newRow.push(student.nilai);
            else newRow.push("");
          }
          barisBaru.push(newRow);
        }
      }

      // Tulis Ulang Semua Perubahan Sekaligus (Batch Update)
      if (data.length > 1) {
        sheet.getRange(1, 1, data.length, data[0].length).setValues(data);
      }

      // Tambahkan Baris Baru Sekaligus (Batch Insert)
      if (barisBaru.length > 0) {
        sheet.getRange(sheet.getLastRow() + 1, 1, barisBaru.length, barisBaru[0].length).setValues(barisBaru);
      }

      return responSukses({ message: 'Nilai mata pelajaran berhasil diproses!' });
    }
  } catch (err) {
    return responError('Gagal memproses nilai: ' + err.message);
  } finally {
    // 2. Wajib melepaskan sistem antrean agar tidak macet
    lock.releaseLock(); 
  }
}

// 6. FUNGSI TARIK DATA NILAI & UPDATE NILAI INDIVIDU
// =================================================================
function getDataNilai(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('NILAI ' + e.parameter.kelas);
  if (!sheet) return responSukses({ data: [], headers: [] });
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return responSukses({ data: [], headers: data[0] || [] });
  var rows = [];
  for (var i = 1; i < data.length; i++) rows.push(data[i]);
  return responSukses({ data: rows, headers: data[0] });
}

function updateDataNilai(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var kelas = e.parameter.kelas; var nis = e.parameter.nis;
  var newData = JSON.parse(e.parameter.data_nilai);
  var sheet = ss.getSheetByName('NILAI ' + kelas);
  if (!sheet) return responError("Sheet tidak ditemukan.");
  var data = sheet.getDataRange().getValues();
  var headers = data[0]; var rowIndex = -1;
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0].toString() === nis.toString() || data[i][0].toString() === "'" + nis) { rowIndex = i + 1; break; }
  }
  if (rowIndex === -1) return responError("Data tidak ditemukan.");
  
  // 1. Tulis nilai yang diedit ke sheet
  for(var key in newData) {
    var colIndex = headers.indexOf(key);
    if (colIndex > -1) {
       sheet.getRange(rowIndex, colIndex + 1).setValue(newData[key]);
    }
  }
  
  // 2. Baca ulang seluruh baris untuk menghitung Total yang akurat anti-salah
  var currentRowData = sheet.getRange(rowIndex, 1, 1, headers.length).getValues()[0];
  var total = 0; 
  var totalMapel = headers.length - 5;
  
  if (kelas.includes('TK')) {
      var count = 0;
      // --- KODE PERBAIKAN: AMBIL DARI KOLOM N1, N2, N3 SECARA SPESIFIK ---
      var idxN1 = headers.findIndex(h => h.toString().toLowerCase() === 'nilai 1' || h.toString().toLowerCase() === 'n1');
      var idxN2 = headers.findIndex(h => h.toString().toLowerCase() === 'nilai 2' || h.toString().toLowerCase() === 'n2');
      var idxN3 = headers.findIndex(h => h.toString().toLowerCase() === 'nilai 3' || h.toString().toLowerCase() === 'n3');
      
      [idxN1, idxN2, idxN3].forEach(function(c) {
          if (c > -1) {
              var val = currentRowData[c];
              if (val !== "" && !isNaN(val)) { total += parseFloat(val); count++; }
          }
      });
      var rata = count > 0 ? (total/count).toFixed(1) : 0;
      // -----------------------------------------------------------
  } else {
      for (var h = 5; h < headers.length; h++) {
         var val = currentRowData[h];
         if (val !== "" && !isNaN(val)) { total += parseFloat(val); }
      }
      var rata = totalMapel > 0 ? (total/totalMapel).toFixed(1) : 0; // Dibagi jumlah kolom mapel
  }
  
  var totalCol = headers.indexOf('Total Nilai');
  var rataCol = headers.indexOf('Rata-rata');
  if (totalCol > -1) sheet.getRange(rowIndex, totalCol + 1).setValue(total);
  if (rataCol > -1) sheet.getRange(rowIndex, rataCol + 1).setValue(rata);
  
  return responSukses({message: "Data nilai berhasil diperbarui."});
}

function responSukses(dataObj) { dataObj.status = 'success'; return ContentService.createTextOutput(JSON.stringify(dataObj)).setMimeType(ContentService.MimeType.JSON); }
function responError(pesan) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: pesan })).setMimeType(ContentService.MimeType.JSON); }

// 8. FUNGSI RANKING KELAS (DIPERBARUI UNTUK AGREGASI TK)
// =================================================================
function getRankingKelas(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var kelasPilih = e.parameter.kelas;
  
  var sheetSantri = ss.getSheetByName('SANTRI');
  var dataSantri = sheetSantri ? sheetSantri.getDataRange().getValues() : [];
  var santriMap = {};
  for (var i = 1; i < dataSantri.length; i++) {
    var nisBersih = dataSantri[i][0].toString().replace(/'/g, "").trim();
    santriMap[nisBersih] = { jk: dataSantri[i][2] || '-', ttl: dataSantri[i][3] || '-', alamat: dataSantri[i][5] || '-', ayah: dataSantri[i][6] || '-', ibu: dataSantri[i][7] || '-' };
  }

  var sheetNilai = ss.getSheetByName('NILAI ' + kelasPilih);
  if (!sheetNilai) return responSukses({ data: [] });
  var dataNilai = sheetNilai.getDataRange().getValues();
  if (dataNilai.length <= 1) return responSukses({ data: [] });
  var headers = dataNilai[0];
  var totalCol = headers.indexOf('Total Nilai');
  var rataCol = headers.indexOf('Rata-rata');
  var namaCol = headers.indexOf('Nama Lengkap');
  var nisCol = headers.indexOf('NIS');
  
  var agregasiRanking = {};
  
  for (var i = 1; i < dataNilai.length; i++) {
    var nisStr = dataNilai[i][nisCol].toString().replace(/'/g, "").trim();
    var totalRow = parseFloat(dataNilai[i][totalCol]) || 0;
    var rataRow = parseFloat(dataNilai[i][rataCol]) || 0;
    
    if (!agregasiRanking[nisStr]) {
        var biodata = santriMap[nisStr] || {jk: '-', ttl: '-', ayah: '-', ibu: '-', alamat: '-'};
        agregasiRanking[nisStr] = {
            nis: nisStr, nama: dataNilai[i][namaCol], total: 0, rata: 0, hitungBaris: 0,
            jk: biodata.jk, ttl: biodata.ttl, ayah: biodata.ayah, ibu: biodata.ibu, alamat: biodata.alamat
        };
    }
    agregasiRanking[nisStr].total += totalRow;
    agregasiRanking[nisStr].rata += rataRow;
    agregasiRanking[nisStr].hitungBaris++;
  }
  
 var rankingList = [];
  for (var key in agregasiRanking) {
     var r = agregasiRanking[key];
     if (kelasPilih.includes('TK')) {
         r.rata = (r.total / r.hitungBaris).toFixed(1);
     } else {
         r.rata = r.rata.toFixed(1); 
     }
     rankingList.push(r);
  }
  
  // --- KODE LOGIKA RANKING KEMBAR ---
  rankingList.sort(function(a, b) { 
      if (b.total !== a.total) {
          return b.total - a.total; 
      }
      return parseFloat(b.rata) - parseFloat(a.rata);
  });
  
  var rankAktual = 1;
  for (var k = 0; k < rankingList.length; k++) { 
      if (k > 0 && rankingList[k].total === rankingList[k-1].total && rankingList[k].rata === rankingList[k-1].rata) {
          rankingList[k].rank = rankAktual; 
      } else {
          rankAktual = k + 1; 
          rankingList[k].rank = rankAktual;
      }
  }
  // ----------------------------------

  return responSukses({ data: rankingList });
}

// 9. FUNGSI BINTANG PELAJAR (LENGKAP DENGAN JENIS KELAMIN)
// =================================================================
function getBintangPelajar() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  
  // 1. Map Biodata Santri
  var sheetSantri = ss.getSheetByName('SANTRI');
  var dataSantri = sheetSantri ? sheetSantri.getDataRange().getValues() : [];
  var santriMap = {};
  for (var i = 1; i < dataSantri.length; i++) {
    var nisBersih = dataSantri[i][0].toString().replace(/'/g, "").trim();
    santriMap[nisBersih] = {
      jk: dataSantri[i][2] || '-',     
      ttl: dataSantri[i][3] || '-',     
      alamat: dataSantri[i][5] || '-',  
      ayah: dataSantri[i][6] || '-',    
      ibu: dataSantri[i][7] || '-'      
    };
  }

  var juaraList = [];

  // 2. Loop ke seluruh sheet mencari awalan "NILAI "
  for (var s = 0; s < sheets.length; s++) {
    var sheetName = sheets[s].getName();
    if (sheetName.indexOf('NILAI ') === 0) { 
      var dataNilai = sheets[s].getDataRange().getValues();
      if (dataNilai.length > 1) {
        var headers = dataNilai[0];
        var totalCol = headers.indexOf('Total Nilai');
        var rataCol = headers.indexOf('Rata-rata');
        var namaCol = headers.indexOf('Nama Lengkap');
        var nisCol = headers.indexOf('NIS');
        var kelasVal = sheetName.replace('NILAI ', '');
        
        // --- KODE PERBAIKAN: AGREGASI BINTANG PELAJAR UNTUK TK ---
        var agregasiKelas = {};
        for (var i = 1; i < dataNilai.length; i++) {
          var nisStr = dataNilai[i][nisCol].toString().replace(/'/g, "").trim();
          
          if (!agregasiKelas[nisStr]) {
              agregasiKelas[nisStr] = {
                  nis: nisStr, nama: dataNilai[i][namaCol], total: 0, rata: 0, hitungBaris: 0, kelas: kelasVal
              };
          }
          agregasiKelas[nisStr].total += parseFloat(dataNilai[i][totalCol]) || 0;
          agregasiKelas[nisStr].rata += parseFloat(dataNilai[i][rataCol]) || 0;
          agregasiKelas[nisStr].hitungBaris++;
        }
        
        var kelasRanking = [];
        for (var key in agregasiKelas) {
            var r = agregasiKelas[key];
            if (kelasVal.includes('TK')) {
                r.rata = r.hitungBaris > 0 ? (r.total / r.hitungBaris).toFixed(1) : 0;
            } else {
                r.rata = r.rata.toFixed(1);
            }
            kelasRanking.push(r);
        }
        
        // Cari juara 1 di kelas ini berdasarkan Total Nilai. 
        // Jika Total Nilai kembar, pemenangnya ditentukan oleh Rata-rata terbesar.
        kelasRanking.sort(function(a, b) { 
            if (b.total !== a.total) {
                return b.total - a.total; 
            }
            return parseFloat(b.rata) - parseFloat(a.rata);
        });
        
        
        if (kelasRanking.length > 0) {
          // Ambil patokan nilai Juara 1 (Nilai Tertinggi di kelas tersebut)
          var topTotal = kelasRanking[0].total;
          var topRata = kelasRanking[0].rata;
          
          // Lakukan perulangan untuk mengecek semua anak di kelas tersebut
          for (var j = 0; j < kelasRanking.length; j++) {
              // Jika nilai anak ini SAMA PERSIS dengan nilai patokan Juara 1, masukkan ke daftar
              if (kelasRanking[j].total === topTotal && kelasRanking[j].rata === topRata) {
                  var juara = kelasRanking[j];
                  var biodata = santriMap[juara.nis] || {jk: '-', ttl: '-', ayah: '-', ibu: '-', alamat: '-'};
                  
                  // Penegas nilai rank agar di web otomatis tertulis angka 1 untuk semua juara kembar
                  juara.rank = 1; 
                  
                  juara.jk = biodata.jk;
                  juara.ttl = biodata.ttl; 
                  juara.ayah = biodata.ayah; 
                  juara.ibu = biodata.ibu; 
                  juara.alamat = biodata.alamat;
                  
                  juaraList.push(juara);
              } else {
                  // Jika nilainya sudah lebih kecil dari Juara 1, hentikan pencarian di kelas ini
                  break; 
              }
          }
        }
      }
    }
  }
  
  return responSukses({ data: juaraList });
}

function simpanPengaturan(e) {

       sheetSet = ss.insertSheet('SETTING_RAPOR');
       sheetSet.appendRow(['Kelas', 'Semester', 'Tahun', 'Tanggal', 'Wali', 'Kepala', 'URL Wali', 'URL Kepala', 'URL Stempel', 'Status Rilis']);
    }
    
    var dSet = sheetSet.getDataRange().getValues();
    var rowIndexSet = -1;
    for (var i = 1; i < dSet.length; i++) {
      if (dSet[i][0] == kelas) { rowIndexSet = i + 1; break; }
    }
    
    if (rowIndexSet === -1) {
      sheetSet.appendRow([kelas, setUmum.semester, setUmum.tahun, setUmum.tanggal, setUmum.wali, setUmum.kepala, '', '', '']);
    } else {

      sheetSet.getRange(rowIndexSet, 4).setValue(setUmum.tanggal);
      sheetSet.getRange(rowIndexSet, 5).setValue(setUmum.wali);
      sheetSet.getRange(rowIndexSet, 6).setValue(setUmum.kepala);
      sheetSet.getRange(rowIndexSet, 10).setValue(setUmum.status_rilis || 'Sembunyi');
    }

    // 2. Simpan Master Mapel & Kamus ke MASTER_MAPEL
    var mapelTulis = e.parameter.mapel_tulis;
    var mapelPraktek = e.parameter.mapel_praktek;
    var mapelBaca = e.parameter.mapel_baca;
    var kamus = e.parameter.kamus;

    var sheetMapel = ss.getSheetByName('MASTER_MAPEL');
    if (!sheetMapel) {
        sheetMapel = ss.insertSheet('MASTER_MAPEL');
        sheetMapel.appendRow(['Kelas', 'Tulis', 'Praktek', 'Baca', 'Kamus']);
    }
    var dMapel = sheetMapel.getDataRange().getValues();
    var rowMapel = -1;
    for (var m = 1; m < dMapel.length; m++) {
        if (dMapel[m][0] == kelas) { rowMapel = m + 1; break; }
    }
    
    if (rowMapel === -1) {
        sheetMapel.appendRow([kelas, mapelTulis, mapelPraktek, mapelBaca, kamus]);
    } else {
        sheetMapel.getRange(rowMapel, 2).setValue(mapelTulis);
        sheetMapel.getRange(rowMapel, 3).setValue(mapelPraktek);
        sheetMapel.getRange(rowMapel, 4).setValue(mapelBaca);
        sheetMapel.getRange(rowMapel, 5).setValue(kamus);
    }
    
    // 3. Simpan Detail Santri (Absensi, Kepribadian) ke DETAIL_RAPOR
    var detSantri = JSON.parse(e.parameter.det_santri);
    var sheetDet = ss.getSheetByName('DETAIL_RAPOR');
    if (!sheetDet) {
       sheetDet = ss.insertSheet('DETAIL_RAPOR');
       sheetDet.appendRow(['NIS', 'Kelas', 'Akhlaq', 'Kerajinan', 'Disiplin', 'Rapi', 'Sakit', 'Izin', 'Alpa', 'Catatan', 'Keputusan']);
    }
    var dDet = sheetDet.getDataRange().getValues();
    
    for (var s = 0; s < detSantri.length; s++) {
       var ds = detSantri[s];
       var rowDet = -1;
       for (var j = 1; j < dDet.length; j++) {
         if (dDet[j][0].toString() == ds.nis.toString() && dDet[j][1] == kelas) { rowDet = j + 1; break; }
       }
       
       if (rowDet === -1) {
         sheetDet.appendRow(["'" + ds.nis, kelas, ds.akhlaq, ds.kerajinan, ds.disiplin, ds.rapi, ds.sakit, ds.izin, ds.alpa, ds.catatan, ds.keputusan]);
       } else {
         sheetDet.getRange(rowDet, 3).setValue(ds.akhlaq);
         sheetDet.getRange(rowDet, 4).setValue(ds.kerajinan);
         sheetDet.getRange(rowDet, 5).setValue(ds.disiplin);
         sheetDet.getRange(rowDet, 6).setValue(ds.rapi);
         sheetDet.getRange(rowDet, 7).setValue(ds.sakit);
         sheetDet.getRange(rowDet, 8).setValue(ds.izin);
         sheetDet.getRange(rowDet, 9).setValue(ds.alpa);
         sheetDet.getRange(rowDet, 10).setValue(ds.catatan);
         sheetDet.getRange(rowDet, 11).setValue(ds.keputusan);
       }
    }
    
    return responSukses({ message: 'Pengaturan Rapor Kelas ' + kelas + ' berhasil disimpan!' });
  } catch(error) {
    return responError('Gagal menyimpan: ' + error.message);
  }
}

// 11. MESIN UPLOAD GAMBAR KE GOOGLE DRIVE
// =================================================================
function uploadGambarDrive(e) {
  try {
    var kelas = e.parameter.kelas;
    var jenis = e.parameter.jenis; // 'wali', 'kepala', 'stempel'
    var base64Data = e.parameter.data;

    // 1. Ekstrak data base64
    var cleanBase64 = base64Data.split(',')[1];
    var decoded = Utilities.base64Decode(cleanBase64);
    var fileName = jenis.toUpperCase() + "_" + kelas + ".png"; // Nama otomatis
    var blob = Utilities.newBlob(decoded, 'image/png', fileName);

    // 2. Buat atau Cari Folder Khusus di Drive
    var folderName = "ASET_RAPOR_MADRASAH";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    // 3. Hapus file lama jika ada (Agar Drive tidak penuh)
    var existingFiles = folder.getFilesByName(fileName);
    while (existingFiles.hasNext()) { existingFiles.next().setTrashed(true); }

    // 4. Simpan File Baru
    var file = folder.createFile(blob);
    
    // Pastikan hak akses file langsung dibuka untuk publik (Mencegah delay izin folder)
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    
    // Gunakan API Thumbnail Drive yang kebal dari pemblokiran keamanan (CORS)
    var fileUrl = "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w1000";

    // 5. Catat URL ke Database (Kolom G, H, I)
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetSet = ss.getSheetByName('SETTING_RAPOR');
    var dSet = sheetSet.getDataRange().getValues();
    var rowIndex = -1;
    for(var i=1; i<dSet.length; i++) { if(dSet[i][0] == kelas) { rowIndex = i + 1; break; } }

    if (rowIndex === -1) {
      sheetSet.appendRow([kelas, '', '', '', '', '', '', '', '']);
      rowIndex = sheetSet.getLastRow();
    }

    if (jenis === 'wali') sheetSet.getRange(rowIndex, 7).setValue(fileUrl);
    else if (jenis === 'kepala') sheetSet.getRange(rowIndex, 8).setValue(fileUrl);
    else if (jenis === 'stempel') sheetSet.getRange(rowIndex, 9).setValue(fileUrl);

    return responSukses({ url: fileUrl, message: 'File ' + fileName + ' berhasil disimpan ke Drive!' });
  } catch(err) {
    return responError('Gagal Upload: ' + err.toString());
  }
}

function getRaporTK(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var inputNis = e.parameter.nis;
  if (!inputNis) return responError("NIS tidak diberikan.");

  var nisBersih = inputNis.toString().replace(/'/g, "").trim();

  // 1. Cari Biodata Santri
  var sheetSantri = ss.getSheetByName('SANTRI');
  if (!sheetSantri) return responError("Database Santri tidak ditemukan.");
  var dataSantri = sheetSantri.getDataRange().getValues();
  var biodata = null;
  var kelasSantri = "";

  for (var i = 1; i < dataSantri.length; i++) {
    var nSheet = dataSantri[i][0].toString().replace(/'/g, "").trim();
    if (nSheet === nisBersih) {
      kelasSantri = dataSantri[i][4]; 
      biodata = {
        nis: nisBersih,
        nama: dataSantri[i][1],
        alamat: dataSantri[i][5] || '-',
        kelas: kelasSantri
      };
      break;
    }
  }

  if (!biodata) return responError("Santri dengan NIS " + inputNis + " tidak ditemukan.");

  // 2. Cari Pengaturan Umum & Detail Rapor
  var sheetSet = ss.getSheetByName('SETTING_RAPOR');
  var setUmum = { semester: "-", tahun: "-", tanggal: "-", wali: "-", kepala: "-", url_wali: "", url_kepala: "", url_stempel: "" };
  if (sheetSet) {
    var dSet = sheetSet.getDataRange().getValues();
    for (var i = 1; i < dSet.length; i++) {
      if (dSet[i][0] == kelasSantri) {
        setUmum = {
          semester: dSet[i][1] || "-", tahun: dSet[i][2] || "-", tanggal: dSet[i][3] || "-",
          wali: dSet[i][4] || "-", kepala: dSet[i][5] || "-",
          url_wali: dSet[i][6] || "", url_kepala: dSet[i][7] || "", url_stempel: dSet[i][8] || ""
        };
        break;
      }
    }
  }

  var sheetDet = ss.getSheetByName('DETAIL_RAPOR');
  var detRapor = { akhlaq: "-", kerajinan: "-", disiplin: "-", rapi: "-", sakit: "0", izin: "0", alpa: "0", catatan: "-", keputusan: "-" };
  if (sheetDet) {
    var dDet = sheetDet.getDataRange().getValues();
    for (var i = 1; i < dDet.length; i++) {
      var nDet = dDet[i][0].toString().replace(/'/g, "").trim();
      if (nDet === nisBersih && dDet[i][1] == kelasSantri) {
        detRapor = {
          akhlaq: dDet[i][2] || "-", kerajinan: dDet[i][3] || "-", disiplin: dDet[i][4] || "-", rapi: dDet[i][5] || "-",
          sakit: dDet[i][6] || "0", izin: dDet[i][7] || "0", alpa: dDet[i][8] || "0",
          catatan: dDet[i][9] || "-", keputusan: dDet[i][10] || "-"
        };
        break;
      }
    }
  }

  // 3. Kumpulkan Nilai TK dari semua Hari
  var sheetNilai = ss.getSheetByName('NILAI ' + kelasSantri);
  var daftarNilai = [];
  if (sheetNilai) {
    var dataNilai = sheetNilai.getDataRange().getValues();
    for (var i = 1; i < dataNilai.length; i++) {
      var nNilai = dataNilai[i][0].toString().replace(/'/g, "").trim();
      if (nNilai === nisBersih) {
        if (dataNilai[i][6]) daftarNilai.push({ mapel: dataNilai[i][6], nilai: dataNilai[i][7] || 0 });
        if (dataNilai[i][8]) daftarNilai.push({ mapel: dataNilai[i][8], nilai: dataNilai[i][9] || 0 });
        if (dataNilai[i][10]) daftarNilai.push({ mapel: dataNilai[i][10], nilai: dataNilai[i][11] || 0 });
      }
    }
  }

  return responSukses({ biodata: biodata, pengaturan: setUmum, detail: detRapor, nilai: daftarNilai });
}

// FUNGSI IMPORT SANTRI MASSAL (EXCEL/CSV) - VERSI UPSERT & ANTI HILANG 0 / +
// =================================================================
function importSantriBulk(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('SANTRI');
    var dataMentah = JSON.parse(e.parameter.data_import);
    
    // Hapus baris pertama jika itu adalah header (judul kolom)
    if (dataMentah.length > 0 && isNaN(parseInt(dataMentah[0][0]))) {
      dataMentah.shift(); 
    }
    
    if (dataMentah.length === 0) {
      return responError('File kosong atau format salah.');
    }
    
    // 1. Tarik data yang sudah ada di sheet untuk dicocokkan
    var existingData = sheet.getDataRange().getValues();
    var nisMap = {};
    
    for (var i = 1; i < existingData.length; i++) {
      var nisLama = existingData[i][0].toString().replace(/'/g, "").trim();
      nisMap[nisLama] = i + 1; 
    }
    
    var dataBaruDitambah = [];
    var jumlahDiupdate = 0;
    
    // 2. Proses data yang di-import satu per satu
    for (var j = 0; j < dataMentah.length; j++) {
      var barisImport = dataMentah[j];
      var nisImport = barisImport[0].toString().replace(/'/g, "").trim();
      
      // KUNCI PERBAIKAN: Amankan format NIS agar 0 di depan tidak hilang
      barisImport[0] = "'" + nisImport;
      
      // KUNCI PERBAIKAN: Amankan kolom No HP (Kolom ke-9 / Indeks 8)
      if (barisImport.length > 8 && barisImport[8] !== null && barisImport[8] !== undefined) {
          // Bersihkan jika ada tanda petik bawaan dari CSV, lalu ambil teks aslinya
          var hpAsli = barisImport[8].toString().replace(/^'/, "").trim();
          
          // Suntikkan tanda petik tunggal (') di paling depan
          // Google Sheets akan membaca ini sebagai Teks. Jadi 08..., +62..., +60... akan aman.
          barisImport[8] = "'" + hpAsli;
      }
      
      // 3. Cek apakah NIS sudah ada di database (Upsert)
      if (nisMap[nisImport]) {
        var barisTarget = nisMap[nisImport];
        sheet.getRange(barisTarget, 1, 1, barisImport.length).setValues([barisImport]);
        jumlahDiupdate++;
      } else {
        dataBaruDitambah.push(barisImport);
      }
    }
    
    // 4. Tambahkan data yang benar-benar baru sekaligus di baris paling bawah
    if (dataBaruDitambah.length > 0) {
      var lastRow = sheet.getLastRow() + 1;
      var jumlahKolom = dataBaruDitambah[0].length;
      sheet.getRange(lastRow, 1, dataBaruDitambah.length, jumlahKolom).setValues(dataBaruDitambah);
    }
    
    return responSukses({
      message: 'Import Selesai! ' + jumlahDiupdate + ' data diperbarui & ' + dataBaruDitambah.length + ' santri baru ditambahkan.'
    });
    
  } catch (err) {
    return responError('Gagal import: ' + err.message);
  }
}

// FUNGSI SPP (SIMPAN & UPDATE) - OPTIMASI BATCH & ANTI-TABRAKAN
// =================================================================
function saveSppData(e) {
  // 1. Lampu Lalu Lintas untuk Keuangan (Wajib Antre)
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    return responError('Sistem keuangan sedang digunakan admin lain. Silakan coba lagi.');
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('SPP');
    
    if (!sheet) {
      sheet = ss.insertSheet('SPP');
      sheet.appendRow(['NIS', 'Nama Lengkap', 'Kelas', 'Keterangan Pembayaran', 'Nominal Bayar', 'Status']);
      sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#d1fae5");
    }
    
    var nis = e.parameter.nis;
    var nama = e.parameter.nama;
    var kelas = e.parameter.kelas;
    var keterangan = e.parameter.keterangan;
    var nominal = e.parameter.nominal; 
    var status = e.parameter.status;
    
    var data = sheet.getDataRange().getValues();
    var isUpdated = false;
    
    // 2. Cek dan Update di dalam Memori Array
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] == nis && data[i][3] == keterangan) {
        data[i][4] = nominal; 
        data[i][5] = status;  
        isUpdated = true;
        break;
      }
    }
    
    // 3. Eksekusi Penulisan (Tulis sekaligus jika ada update, atau tambah baris jika baru)
    if (isUpdated) {
      sheet.getRange(1, 1, data.length, data[0].length).setValues(data);
    } else {
      sheet.appendRow(["'" + nis, nama, kelas, "'" + keterangan, nominal, status]);
    }
    
    return responSukses({ message: 'Data transaksi SPP berhasil disimpan!' });

  } catch (err) {
    return responError('Gagal menyimpan SPP: ' + err.message);
  } finally {
    // 4. Buka kembali gembok antrean
    lock.releaseLock();
  }
}

// 1. FUNGSI SIMPAN ABSENSI BARCODE & MANUAL (ANTI GANDA & ANTI LIBUR)
// =================================================================
function simpanAbsenBarcode(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var timezone = Session.getScriptTimeZone();
  
  // A. CEK HARI JUM'AT DI SERVER (Keamanan Lapis 2)
  var dateWIB = new Date(new Date().toLocaleString("en-US", {timeZone: "Asia/Jakarta"}));
  if (dateWIB.getDay() === 5) return responError("Hari ini Jum'at. Madrasah libur.");

  var formatTanggalHariIni = Utilities.formatDate(dateWIB, timezone, "dd/MM/yyyy");

  // B. CEK SAKELAR LIBUR MANUAL (KODE BARU YANG HARUS ADA)
  var sheetStatus = ss.getSheetByName('STATUS_ABSEN');
  if (sheetStatus && sheetStatus.getLastRow() >= 2) {
    var statusSistem = sheetStatus.getRange("A2").getValue();
    var ketSistem = sheetStatus.getRange("B2").getValue();
    if (statusSistem === "Libur") {
      return responError('Madrasah sedang LIBUR (' + ketSistem + '). Absensi ditolak!');
    }
  }

  // C. LANJUTKAN PROSES SIMPAN
  var sheet = ss.getSheetByName('Absensi');

  // C. LANJUTKAN PROSES SIMPAN
  var sheet = ss.getSheetByName('Absensi'); 
  if (!sheet) {
    sheet = ss.insertSheet('Absensi');
    sheet.appendRow(['Waktu Absen', 'NIS', 'Nama Santri', 'Kelas', 'Status']);
  }

  var nis = e.parameter.nis;
  var nama = e.parameter.nama;
  var kelas = e.parameter.kelas;
  var status = e.parameter.status;
  var data = sheet.getDataRange().getValues();
  var sudahAbsen = false;

  for (var i = data.length - 1; i > 0; i--) {
    var barisTanggal = data[i][0];
    var barisNis = data[i][1];     
    var nisDiDatabase = barisNis.toString().replace(/'/g, "").trim();
    var nisMasuk = nis.toString().replace(/'/g, "").trim();
    
    if (barisTanggal && nisDiDatabase == nisMasuk) {
      var formatTanggalSheet = Utilities.formatDate(new Date(barisTanggal), timezone, "dd/MM/yyyy");
      if (formatTanggalSheet === formatTanggalHariIni) {
        sudahAbsen = true; break;
      }
    }
  }
  
  if (sudahAbsen) {
    return responError('Afwan, Kakak ' + nama + ' sudah tercatat hari ini.');
  }
  
  // Waktu spesifik server
 var waktuInput = Utilities.formatDate(dateWIB, "GMT+7", "yyyy-MM-dd HH:mm:ss");
  sheet.appendRow([waktuInput, "'" + nis, nama, kelas, status]);
  return responSukses({ message: 'Kehadiran berhasil disimpan' });
}

// 2. FUNGSI UNTUK MENGAMBIL BULAN YANG TERSEDIA (VERSI BERSIH)
// =================================================================
function getBulanTersedia(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Absensi');
  if (!sheet) return responSukses({ data: [] });

  var data = sheet.getDataRange().getValues();
  var bulanUnik = {};
  var namaBulanIndo = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];

  for (var i = 1; i < data.length; i++) {
    var cellWaktu = data[i][0];
    if (!cellWaktu) continue; 
    
    // Cukup gunakan pembaca waktu standar karena datanya sudah rapi
    var tgl = new Date(cellWaktu);
    
    if (!isNaN(tgl.getTime())) {
      var bulan = tgl.getMonth();
      var tahun = tgl.getFullYear();
      var valueBulan = tahun + "-" + (bulan + 1 < 10 ? '0' : '') + (bulan + 1); 
      var labelBulan = namaBulanIndo[bulan] + " " + tahun; 
      if (!bulanUnik[valueBulan]) bulanUnik[valueBulan] = labelBulan;
    }
  }

  var result = [];
  for (var key in bulanUnik) result.push({ value: key, label: bulanUnik[key] });
  result.sort(function(a, b) { return b.value.localeCompare(a.value); });
  return responSukses({ data: result });
}

// 3. FUNGSI UNTUK MENARIK DATA REKAPITULASI (VERSI BERSIH)
// =================================================================
function getRekapBulanan(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var blnTahunReq = e.parameter.bulan_tahun; 
  var kelasReq = e.parameter.kelas;

  var sheetAbsen = ss.getSheetByName('Absensi');
  if (!sheetAbsen) return responError("Sheet 'Absensi' tidak ditemukan.");
  
  var dataAbsen = sheetAbsen.getDataRange().getValues();
  var rekapMap = {};

  for (var j = 1; j < dataAbsen.length; j++) {
    var cellWaktu = dataAbsen[j][0]; 
    if (!cellWaktu) continue; 
    
    var nis = dataAbsen[j][1].toString().replace(/'/g, "").trim();
    var nama = dataAbsen[j][2];
    var kelas = dataAbsen[j][3];
    var status = dataAbsen[j][4];

    var kelasBersih = kelas.toString().trim();
    var kelasReqBersih = kelasReq.toString().trim();
    var statusBersih = status.toString().trim();

    if (kelasBersih === kelasReqBersih) {
      
      // Pembacaan waktu natural
      var tglObj = new Date(cellWaktu);

      if (!isNaN(tglObj.getTime())) {
          var blnAngka = tglObj.getMonth() + 1;
          var blnStr = blnAngka < 10 ? '0' + blnAngka : blnAngka.toString();
          var thn = tglObj.getFullYear();
          var blnTahunData = thn + "-" + blnStr;

          if (blnTahunData === blnTahunReq) {
            if (!rekapMap[nis]) {
              rekapMap[nis] = { nis: nis, nama: nama, hadir: 0, izin: 0, sakit: 0, alfa: 0 };
            }
            if (statusBersih === "Hadir") rekapMap[nis].hadir++;
            else if (statusBersih === "Izin") rekapMap[nis].izin++;
            else if (statusBersih === "Sakit") rekapMap[nis].sakit++;
            else if (statusBersih === "Alfa") rekapMap[nis].alfa++;
          }
      }
    }
  }

  var resultArray = Object.keys(rekapMap).map(function(k){ return rekapMap[k]; });
  resultArray.sort(function(a, b) { return a.nama.localeCompare(b.nama); });
  return responSukses({ data: resultArray });
}

function setLiburRentang(e) {

    sheetLibur = ss.insertSheet('HARI_LIBUR');

  var keterangan = e.parameter.keterangan;

  akhir.setHours(0, 0, 0, 0);

  var currentDate = new Date(mulai.getTime());

  while (currentDate <= akhir) {
    var tglFormat = Utilities.formatDate(currentDate, "GMT+7", "dd/MM/yyyy");

    for (var i = 1; i < dataLama.length; i++) {
       if (dataLama[i][0].toString().trim() === tglFormat) { sudahAda = true; break; }
    }
    
    if (!sudahAda) sheetLibur.appendRow(["'" + tglFormat, keterangan]);
    currentDate.setDate(currentDate.getDate() + 1); 
  }
  return responSukses({message: "Jadwal libur ditambahkan."});
}

// FUNGSI ROBOT AUTO-ALFA (BERJALAN SETIAP SORE)
// =================================================================
function autoTutupAbsenHariIni() {
  var dateWIB = new Date(new Date().toLocaleString("en-US", {timeZone: "Asia/Jakarta"}));
  
  // 1. CEK JUM'AT (Otomatis tidur)
  if (dateWIB.getDay() === 5) return; 

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hariIni = Utilities.formatDate(dateWIB, "GMT+7", "dd/MM/yyyy");

  // 2. CEK SAKELAR LIBUR MANUAL (Robot otomatis tidur jika statusnya "Libur")
  var sheetStatus = ss.getSheetByName('STATUS_ABSEN');
  if (sheetStatus && sheetStatus.getLastRow() >= 2) {
    if (sheetStatus.getRange("A2").getValue() === "Libur") return;
  }

  var sheetSantri = ss.getSheetByName('SANTRI');
  var sheetAbsen = ss.getSheetByName('Absensi'); 
  if (!sheetSantri || !sheetAbsen) return;

  var dataAbsen = sheetAbsen.getDataRange().getValues();
  var absenHariIni = {};
  var jumlahHadirHariIni = 0;
  
  // 3. REKAP YANG SUDAH ABSEN (Hadir/Izin/Sakit)
  for (var j = 1; j < dataAbsen.length; j++) {
     var cellWaktu = dataAbsen[j][0];
     if (!cellWaktu) continue;

     var tglAbsen = Utilities.formatDate(new Date(cellWaktu), "GMT+7", "dd/MM/yyyy");
     var nisAbsen = dataAbsen[j][1].toString().replace(/'/g, "").trim();
     if (tglAbsen === hariIni) {
        absenHariIni[nisAbsen] = true; 
        jumlahHadirHariIni++;
     }
  }

  // 4. DETEKSI LIBUR MENDADAK
  // Jika sampai jam 18.00 kosong total tidak ada yang absen 1 orang pun, berarti sekolah batal buka.
  // Jangan Alfa-kan massal!
  if (jumlahHadirHariIni === 0) return; 

  // PERBAIKAN: Gunakan format standar ISO agar sinkron dengan scanner
  var waktuTutup = Utilities.formatDate(dateWIB, "GMT+7", "yyyy-MM-dd HH:mm:ss");
  
  var dataSantri = sheetSantri.getDataRange().getValues();
  // 5. EKSEKUSI ALFA UNTUK YANG TIDAK ADA DI DAFTAR
  for (var i = 1; i < dataSantri.length; i++) {
     var nis = dataSantri[i][0].toString().replace(/'/g, "").trim();
     var nama = dataSantri[i][1];
     var kelas = dataSantri[i][4]; 
     
     if (!absenHariIni[nis]) {
        sheetAbsen.appendRow([ waktuTutup, "'" + nis, nama, kelas, "Alfa" ]);
     }
  }
}

// FUNGSI BUKU KAS (MENGHITUNG SALDO & MENCATAT PENGELUARAN)
// =================================================================

function getBukuKas() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Hitung Total Pemasukan SPP
  var sheetSPP = ss.getSheetByName('SPP');
  var totalMasuk = 0;
  if (sheetSPP) {
    var dataMasuk = sheetSPP.getDataRange().getValues();
    for (var i = 1; i < dataMasuk.length; i++) {
      totalMasuk += parseFloat(dataMasuk[i][4]) || 0; // Kolom ke-5 adalah Nominal
    }
  }
  
  // 2. Hitung Total Pengeluaran
  var sheetKeluar = ss.getSheetByName('PENGELUARAN');
  var totalKeluar = 0;
  if (sheetKeluar) {
    var dataKeluar = sheetKeluar.getDataRange().getValues();
    for (var j = 1; j < dataKeluar.length; j++) {
      totalKeluar += parseFloat(dataKeluar[j][2]) || 0; // Kolom ke-3 adalah Nominal
    }
  }
  
  // 3. Kalkulasi Saldo
  var saldoAkhir = totalMasuk - totalKeluar;
  
  return responSukses({ 
    masuk: totalMasuk, 
    keluar: totalKeluar, 
    saldo: saldoAkhir 
  });
}

function addPengeluaran(e) {
  try {
    var nominalReq = parseFloat(e.parameter.nominal) || 0;
    
    // [KEAMANAN SERVER] Cek saldo secara live di backend sebelum memproses
    // Mencegah tembusan hacker jika mencoba mem-bypass peringatan di Frontend
    var dataKas = JSON.parse(getBukuKas().getContent());
    if (nominalReq > dataKas.saldo) {
       return responError("Transaksi ditolak! Saldo saat ini hanya tersisa Rp " + dataKas.saldo);
    }
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('PENGELUARAN');
    
    // Buat Sheet Pengeluaran otomatis jika belum ada
    if (!sheet) {
      sheet = ss.insertSheet('PENGELUARAN');
      sheet.appendRow(['Tanggal', 'Keterangan Pengeluaran', 'Nominal Keluar', 'Diinput Oleh', 'Waktu Sistem']);
      sheet.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#fca5a5");
    }
    
    var tgl = e.parameter.tanggal;
    var ket = e.parameter.keterangan;
    var usr = e.parameter.user;
    var wkt = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss");
    
    sheet.appendRow(["'" + tgl, ket, nominalReq, usr, wkt]);
    
    return responSukses({ message: "Pengeluaran sukses dicatat." });
    
  } catch (err) {
    return responError("Gagal memproses: " + err.message);
  }
}

// FUNGSI GABUNGAN BUKU KAS (UNTUK LAPORAN CETAK)
// =================================================================
function getLaporanKas(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var keyword = e.parameter.keyword ? e.parameter.keyword.toString().toLowerCase() : "";
  var jenisReq = e.parameter.jenis; // "Semua", "Masuk", "Keluar"

  var result = [];
  var totMasuk = 0;
  var totKeluar = 0;

  // 1. Tarik Data SPP (Pemasukan)
  if (jenisReq === "Semua" || jenisReq === "Masuk") {
    var sheetSPP = ss.getSheetByName('SPP');
    if (sheetSPP) {
      var dataSPP = sheetSPP.getDataRange().getValues();
      for (var i = 1; i < dataSPP.length; i++) {
        var ket = dataSPP[i][3];
        if (ket instanceof Date) ket = Utilities.formatDate(ket, Session.getScriptTimeZone(), "dd MMMM yyyy");
        else ket = ket.toString();

        var nominal = parseFloat(dataSPP[i][4]) || 0;
        var rincian = "Terima SPP dari " + dataSPP[i][1] + " (" + ket + ")";

        // Cek kecocokan teks
        if (keyword === "" || rincian.toLowerCase().indexOf(keyword) !== -1) {
          result.push({ rincian: rincian, jenis: "Masuk", masuk: nominal, keluar: 0 });
          totMasuk += nominal;
        }
      }
    }
  }

  // 2. Tarik Data Pengeluaran
  if (jenisReq === "Semua" || jenisReq === "Keluar") {
    var sheetKeluar = ss.getSheetByName('PENGELUARAN');
    if (sheetKeluar) {
      var dataKeluar = sheetKeluar.getDataRange().getValues();
      for (var j = 1; j < dataKeluar.length; j++) {
        var tgl = dataKeluar[j][0];
        if (tgl instanceof Date) tgl = Utilities.formatDate(tgl, Session.getScriptTimeZone(), "dd/MM/yyyy");
        else tgl = tgl.toString().replace(/'/g, "");

        var ket = dataKeluar[j][1].toString();
        var nominal = parseFloat(dataKeluar[j][2]) || 0;
        var rincian = tgl + " - " + ket;

        // Cek kecocokan teks
        if (keyword === "" || rincian.toLowerCase().indexOf(keyword) !== -1) {
          result.push({ rincian: rincian, jenis: "Keluar", masuk: 0, keluar: nominal });
          totKeluar += nominal;
        }
      }
    }
  }

  // Balik urutan agar transaksi terbaru ada di paling atas (Z ke A)
  result.reverse();

  return responSukses({ data: result, masuk: totMasuk, keluar: totKeluar });
}

// FUNGSI MENGHAPUS DATA SANTRI DARI DATABASE
// =================================================================
function hapusDataSantriServer(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('SANTRI');
  
  if (!sheet) return responError('Database Santri tidak ditemukan.');
  
  var nisTarget = e.parameter.nis;
  var data = sheet.getDataRange().getValues();
  
  // Looping dari bawah agar indeks baris akurat saat dihapus
  for (var i = data.length - 1; i > 0; i--) {
    var nisData = data[i][0].toString().replace(/'/g, "").trim();
    if (nisData === nisTarget.toString().trim()) {
      sheet.deleteRow(i + 1); // +1 karena array dimulai dari 0
      return responSukses({ message: 'Data santri berhasil dihapus dari sistem.' });
    }
  }
  
  return responError('Data santri tidak ditemukan atau sudah terhapus sebelumnya.');
}

// =================================================================
// FUNGSI HAPUS KAS (PEMASUKAN / PENGELUARAN) DARI BUKU KAS
// =================================================================
function deleteKas(e) {
  // 1. Kunci sistem (Lock Service) agar penghapusan data tidak tabrakan
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(15000)) {
    return responError('Sistem Kas sedang sibuk. Silakan coba lagi.');
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var rincianTarget = e.parameter.rincian; // Teks yang diklik untuk dihapus
    var jenisTarget = e.parameter.jenis;     // 'Masuk' atau 'Keluar'
    
    // --- SKENARIO 1: MENGHAPUS PEMASUKAN (SPP) ---
    if (jenisTarget === 'Masuk') {
      var sheetSPP = ss.getSheetByName('SPP');
      if (!sheetSPP) return responError('Database SPP tidak ditemukan.');
      
      var dataSPP = sheetSPP.getDataRange().getValues();
      var isDeleted = false;
      
      // Loop dari bawah (rekomendasi terbaik saat menghapus baris)
      for (var i = dataSPP.length - 1; i > 0; i--) {
        // Rangkai rincian yang sama dengan yang ditampilkan di laporan kas
        var ket = dataSPP[i][3];
        if (ket instanceof Date) ket = Utilities.formatDate(ket, Session.getScriptTimeZone(), "dd MMMM yyyy");
        else ket = ket.toString();
        
        var rincianData = "Terima SPP dari " + dataSPP[i][1] + " (" + ket + ")";
        
        // Cek kecocokan persis
        if (rincianData === rincianTarget) {
          sheetSPP.deleteRow(i + 1);
          isDeleted = true;
          break; // Hapus satu persatu agar aman
        }
      }
      
      if (isDeleted) return responSukses({ message: 'Pemasukan berhasil dihapus.' });
      else return responError('Data Pemasukan tidak ditemukan.');
    }
    
    // --- SKENARIO 2: MENGHAPUS PENGELUARAN ---
    else if (jenisTarget === 'Keluar') {
      var sheetKeluar = ss.getSheetByName('PENGELUARAN');
      if (!sheetKeluar) return responError('Database Pengeluaran tidak ditemukan.');
      
      var dataKeluar = sheetKeluar.getDataRange().getValues();
      var isDeleted = false;
      
      for (var j = dataKeluar.length - 1; j > 0; j--) {
        // Rangkai rincian yang sama dengan yang ditampilkan di laporan kas
        var tgl = dataKeluar[j][0];
        if (tgl instanceof Date) tgl = Utilities.formatDate(tgl, Session.getScriptTimeZone(), "dd/MM/yyyy");
        else tgl = tgl.toString().replace(/'/g, "");
        
        var ket = dataKeluar[j][1].toString();
        var rincianData = tgl + " - " + ket;
        
        // Cek kecocokan persis
        if (rincianData === rincianTarget) {
          sheetKeluar.deleteRow(j + 1);
          isDeleted = true;
          break;
        }
      }
      
      if (isDeleted) return responSukses({ message: 'Pengeluaran berhasil dihapus.' });
      else return responError('Data Pengeluaran tidak ditemukan.');
    }
    
    else {
       return responError('Jenis transaksi tidak valid.');
    }

  } catch (err) {
    return responError('Terjadi kesalahan sistem: ' + err.message);
  } finally {
    // 2. Lepaskan kunci sistem
    lock.releaseLock();
  }
}
